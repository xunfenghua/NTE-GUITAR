## 范围与入口

- 普通音游模式对应 `PlayMode.RhythmNormal`，UI 发送的模式值为 `rhythm-normal`。地狱模式沿用同一套节奏时钟和判定窗口，并额外判定位置切换。
- 判定流程：`SemiAutoController` 接收全局按键事件 → 按目标音符与输入时间匹配 → `RhythmSessionController` 换算速度比例 → `HitJudgement` 返回 Perfect / Good / Miss → 桥接层将结果发给覆盖层。
- 界面由 `Web/SemiAutoOverlay/index.html`、`styles.css`、`app.js` 组成；WinForms `OverlayWindow` 承载 WebView2，`BridgeController` 与 `BridgeDispatcher` 负责双向通信。

- 主窗体 `UI/MainForm.cs` 还负责创建控制器、指定覆盖层资源地址及挂接主 UI 命令；该文件包含整个应用的大量非音游代码，因此不全文重复，入口位置见下方索引。

```csharp
        string midiPath,
        string extractionStrategy,
        IExistingMidiAdapter midiAdapter,
        IExistingWheelTimingModel wheelTimingModel)
    {
        var targetNotes = midiAdapter.LoadAndSelectTargetNotes(midiPath, extractionStrategy);
        var compiler = new GuitarChartCompiler(wheelTimingModel);
        var chart = compiler.Compile(targetNotes, new ChartCompileOptions
        {
            SourceName = Path.GetFileNameWithoutExtension(midiPath),
            SimultaneousToleranceUs = 20_000,
            UseViterbi = true
        });

        // Preserve the chart and issue list for the UI, export and diagnostics.
        return JsonSerializer.Serialize(chart, new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            WriteIndented = false
        });
    }
}
```

### `SemiAuto/Chart/GuitarChartCompiler.cs`

```csharp
namespace Nte.GuitarChart.Core;

/// <summary>
/// First-pass reference compiler.
/// Input must already have:
/// 1) MIDI ticks converted to absolute microseconds by the existing Tempo Map loader;
/// 2) the user's chosen melody/track filter applied.
/// This compiler does not delete selected notes. Impossible groups remain as Unplayable chart events.
/// </summary>
public sealed class GuitarChartCompiler
{
    private static readonly int[] OpenMidiByPhysicalString = { 0, 64, 59, 55, 50, 45, 40 };

    private sealed record OnsetGroup(IReadOnlyList<ChartNote> Notes, long StartUs);
    private sealed record PositionOption(
        GuitarPosition Position,
        IReadOnlyDictionary<Guid, int> PhysicalStringByNoteId,
        double LocalCost);
    private sealed record PlannedGroup(OnsetGroup Group, IReadOnlyList<PositionOption> Options);

    private readonly IPositionTimingModel _timingModel;

    public GuitarChartCompiler(IPositionTimingModel? timingModel = null)
        => _timingModel = timingModel ?? new WheelPositionTimingModel();

    public GuitarChart Compile(
        IReadOnlyList<RawMidiNote> selectedTargetNotes,
        ChartCompileOptions? options = null)
    {
        options ??= new ChartCompileOptions();

        // Do not silently filter selected target notes.
        var notes = selectedTargetNotes
            .OrderBy(n => n.StartUs)
            .ThenBy(n => n.NoteNumber)
            .Select(n => new ChartNote(
                Guid.NewGuid(),
                Math.Max(0, n.StartUs),
                Math.Max(0, n.DurationUs),
                n.NoteNumber,
                n.Velocity,
                n.Channel,
                n.Program,
                n.TrackIndex,
                n.TrackName,
                null, null, null,
                FeasibilityStatus.Playable))
            .ToList();

        var issues = new List<FeasibilityIssue>();
        foreach (var note in notes.Where(n => n.DurationUs == 0))
        {
            issues.Add(new FeasibilityIssue(
                "ZERO_DURATION",
                IssueSeverity.Warning,
                "音符时值为 0；保留事件，但需检查 MIDI NoteOff 配对。",
                new[] { note.Id },
                note.StartUs));
        }

        var groups = BuildOnsetGroups(notes, options.SimultaneousToleranceUs);
        var planned = groups.Select(g => new PlannedGroup(g, BuildPositionOptions(g, options))).ToList();
        var chosen = SelectGlobalPath(planned, options);
        var resultById = notes.ToDictionary(n => n.Id);
        var selectedGroups = new List<(OnsetGroup Group, PositionOption Option)>();

        for (var i = 0; i < planned.Count; i++)
        {
            var pg = planned[i];
            if (pg.Options.Count == 0 || chosen[i] is null)
            {
                foreach (var note in pg.Group.Notes)
                {
                    resultById[note.Id] = note with
                    {
                        Status = FeasibilityStatus.Unplayable,
                        IssueCode = "NO_COMMON_FRET",
                        IssueMessage = "同一并发组找不到能由当前吉他位置共同演奏的指法；目标音符保留在谱面中。"
                    };
                }

                issues.Add(new FeasibilityIssue(
                    "NO_COMMON_FRET",
                    IssueSeverity.Error,
                    "并发音符无法共用同一品位位置。可尝试分解和弦改编，但必须由用户确认。",
                    pg.Group.Notes.Select(n => n.Id).ToArray(),
                    pg.Group.StartUs));
                continue;
            }

            var option = chosen[i]!;
            selectedGroups.Add((pg.Group, option));

            foreach (var note in pg.Group.Notes)
            {
                var physicalString = option.PhysicalStringByNoteId[note.Id];
                resultById[note.Id] = note with
                {
                    PhysicalStringNo = physicalString,
                    // The game's lane order is opposite physical guitar-string numbering:
                    // physical string 1 maps to game key 6, physical string 6 to game key 1.
                    GameKey = 7 - physicalString,
                    Position = option.Position,
                    Status = FeasibilityStatus.Playable,
                    IssueCode = null,
                    IssueMessage = null
                };
            }
        }

        var positionEvents = BuildPositionEvents(selectedGroups, options, issues);
        foreach (var posEvent in positionEvents.Where(e => e.Status == PositionEventStatus.AtRisk))
        {
            foreach (var note in resultById.Values.Where(n =>
                         n.Position == posEvent.To &&
                         Math.Abs(n.StartUs - posEvent.TargetTimeUs) <= options.SimultaneousToleranceUs))
            {
                if (note.Status == FeasibilityStatus.Playable)
                {
                    resultById[note.Id] = note with
                    {
                        Status = FeasibilityStatus.AtRisk,
                        IssueCode = "POSITION_PREPARE_TIME",
                        IssueMessage = "当前估算的翻页/选槽准备时间不足，需通过真实游戏校准或重新规划验证。"
                    };
                }
            }
        }

        var finalNotes = resultById.Values.OrderBy(n => n.StartUs).ThenBy(n => n.NoteNumber).ToArray();
        return new GuitarChart
        {
            ChartId = Guid.NewGuid().ToString("N"),
            SourceName = options.SourceName,
            Notes = finalNotes,
            PositionEvents = positionEvents,
            Issues = issues,
            DurationUs = finalNotes.Length == 0 ? 0 : finalNotes.Max(n => n.StartUs + n.DurationUs),
            SourceNoteCount = finalNotes.Length,
            PlayableNoteCount = finalNotes.Count(n => n.Status == FeasibilityStatus.Playable),
            AtRiskNoteCount = finalNotes.Count(n => n.Status == FeasibilityStatus.AtRisk),
            UnplayableNoteCount = finalNotes.Count(n => n.Status == FeasibilityStatus.Unplayable)
        };
    }

    private static List<OnsetGroup> BuildOnsetGroups(List<ChartNote> notes, long toleranceUs)
    {
        var groups = new List<OnsetGroup>();
        var cursor = 0;
        while (cursor < notes.Count)
        {
            var firstStart = notes[cursor].StartUs;
            var batch = new List<ChartNote> { notes[cursor] };
            cursor++;
            while (cursor < notes.Count && notes[cursor].StartUs - firstStart <= toleranceUs)
            {
                batch.Add(notes[cursor]);
                cursor++;
            }
            groups.Add(new OnsetGroup(batch, firstStart));
        }
        return groups;
    }

    private static IReadOnlyList<PositionOption> BuildPositionOptions(OnsetGroup group, ChartCompileOptions options)
    {
        var result = new List<PositionOption>();
        for (var fret = 0; fret <= 24; fret++)
        {
            if (!TryAssignDistinctStrings(group.Notes, fret, out var stringMap))
                continue;

            var position = GuitarPosition.FromFret(fret);
            var localCost = Math.Max(0, fret - 12) * options.HighFretPenaltyPerFret;
            foreach (var note in group.Notes)
            {
                var physicalString = stringMap[note.Id];
                if (note.NoteNumber >= 64 && physicalString >= 4) localCost += 0.4;
                if (note.NoteNumber <= 50 && physicalString <= 2) localCost += 0.4;
            }
            result.Add(new PositionOption(position, stringMap, localCost));
        }
        return result;
    }

    private static bool TryAssignDistinctStrings(
        IReadOnlyList<ChartNote> notes, int fret,
        out IReadOnlyDictionary<Guid, int> assignment)
    {
        var ordered = notes.Select(n => new
        {
            Note = n,
            Strings = Enumerable.Range(1, 6)
                .Where(s => OpenMidiByPhysicalString[s] + fret == n.NoteNumber)
                .OrderBy(s => s).ToArray()
        }).OrderBy(x => x.Strings.Length).ToArray();

        var result = new Dictionary<Guid, int>();
        var used = new HashSet<int>();

        bool Search(int index)
        {
            if (index >= ordered.Length) return true;
            var item = ordered[index];
            foreach (var stringNo in item.Strings)
            {
                if (!used.Add(stringNo)) continue;
                result[item.Note.Id] = stringNo;
                if (Search(index + 1)) return true;
                result.Remove(item.Note.Id);
                used.Remove(stringNo);
            }
            return false;
        }

        var success = Search(0);
        assignment = result;
        return success;
    }

    private IReadOnlyList<PositionOption?> SelectGlobalPath(
        IReadOnlyList<PlannedGroup> groups, ChartCompileOptions options)
    {
        var selected = new PositionOption?[groups.Count];
        var activeIndexes = Enumerable.Range(0, groups.Count).Where(i => groups[i].Options.Count > 0).ToArray();
        if (activeIndexes.Length == 0) return selected;

        if (!options.UseViterbi)
        {
            foreach (var i in activeIndexes)
                selected[i] = groups[i].Options.OrderBy(o => o.LocalCost).First();
            return selected;
        }

        var dp = new double[activeIndexes.Length][];
        var back = new int[activeIndexes.Length][];

        for (var layer = 0; layer < activeIndexes.Length; layer++)
        {
            var groupIndex = activeIndexes[layer];
            var candidates = groups[groupIndex].Options;
            dp[layer] = Enumerable.Repeat(double.PositiveInfinity, candidates.Count).ToArray();
            back[layer] = Enumerable.Repeat(-1, candidates.Count).ToArray();

            if (layer == 0)
            {
                for (var j = 0; j < candidates.Count; j++) dp[layer][j] = candidates[j].LocalCost;
                continue;
            }

            var previousGroupIndex = activeIndexes[layer - 1];
            var previousGroup = groups[previousGroupIndex].Group;
            var currentGroup = groups[groupIndex].Group;
            var deltaUs = Math.Max(0, currentGroup.StartUs - previousGroup.StartUs);

            for (var j = 0; j < candidates.Count; j++)
            {
                for (var k = 0; k < groups[previousGroupIndex].Options.Count; k++)
                {
                    var previous = groups[previousGroupIndex].Options[k];
                    var current = candidates[j];
                    var prepareUs = _timingModel.EstimatePrepareUs(previous.Position, current.Position, options);
                    var timingPenalty = deltaUs < prepareUs
                        ? options.InfeasibleTransitionPenalty + (prepareUs - deltaUs) / 1000.0
                        : 0.0;

                    var cost = dp[layer - 1][k]
                               + TransitionCost(previous.Position, current.Position, options)
                               + timingPenalty + current.LocalCost;
                    if (cost < dp[layer][j])
                    {
                        dp[layer][j] = cost;
                        back[layer][j] = k;
                    }
                }
            }
        }

        var lastLayer = activeIndexes.Length - 1;
        var best = Array.IndexOf(dp[lastLayer], dp[lastLayer].Min());
        for (var layer = lastLayer; layer >= 0; layer--)
        {
            selected[activeIndexes[layer]] = groups[activeIndexes[layer]].Options[best];
            best = back[layer][best];
            if (layer > 0 && best < 0) best = 0;
        }
        return selected;
    }

    private static double TransitionCost(GuitarPosition from, GuitarPosition to, ChartCompileOptions options)
    {
        var pageDistance = Math.Abs(to.Page - from.Page);
        var slotDistance = CircularSlotDistance(from.Slot, to.Slot);
        var fretDistance = Math.Abs(to.Fret - from.Fret);
        return pageDistance * options.PageChangeCost
               + slotDistance * options.SlotChangeCost
               + fretDistance * options.FretDistanceCost;
    }

    private static int CircularSlotDistance(int a, int b)
    {
        var direct = Math.Abs(a - b);
        return Math.Min(direct, 6 - direct);
    }

    private IReadOnlyList<PositionEvent> BuildPositionEvents(
        IReadOnlyList<(OnsetGroup Group, PositionOption Option)> groups,
        ChartCompileOptions options,
        List<FeasibilityIssue> issues)
    {
        var events = new List<PositionEvent>();
        if (groups.Count < 2) return events;

        var ordered = groups.OrderBy(g => g.Group.StartUs).ToArray();
        for (var i = 1; i < ordered.Length; i++)
        {
            var previous = ordered[i - 1];
            var current = ordered[i];
            var from = previous.Option.Position;
            var to = current.Option.Position;
            if (from.Page == to.Page && from.Slot == to.Slot) continue;

            var prepareUs = _timingModel.EstimatePrepareUs(from, to, options);
            var availableUs = Math.Max(0, current.Group.StartUs - previous.Group.StartUs);
            var atRisk = availableUs < prepareUs;
            var promptUs = Math.Max(0, current.Group.StartUs - prepareUs);

            events.Add(new PositionEvent(
                Guid.NewGuid(), current.Group.StartUs, promptUs, current.Group.StartUs,
                from, to, PositionSwitchDifficulty.CalculateWeight(from, to),
                atRisk ? PositionEventStatus.AtRisk : PositionEventStatus.Planned,
                atRisk ? "POSITION_PREPARE_TIME" : null));

            if (atRisk)
            {
                issues.Add(new FeasibilityIssue(
                    "POSITION_PREPARE_TIME", IssueSeverity.Warning,
                    $"位置切换 {from} -> {to} 的估算准备时间为 {prepareUs / 1000.0:F0}ms，粗略可用时间为 {availableUs / 1000.0:F0}ms。此为风险提示，不删除音符。",
                    current.Group.Notes.Select(n => n.Id).ToArray(), current.Group.StartUs));
            }
        }
        return events;
    }
}
```

```csharp
                if (i < pageNotes.Length) cuesByNoteIndex.TryGetValue(pageIndex * EventsPerPage + i, out cue);
                if (cue == null) html.Append("<div class=\"cell\"><div class=\"empty-cue\"></div></div>");
                else html.Append("<div class=\"cell\"><div class=\"position-cue\">→ P").Append(cue.To.Page).Append(" / S").Append(cue.To.Slot)
                    .Append("<small>").Append(cue.To.Fret).Append("品　").Append(FormatTime(cue.PromptTimeUs)).Append("</small></div></div>");
            }
            html.Append("</div></div>");

            html.Append("<div class=\"legend\">蓝：可演奏　橙：切换时间风险　紫：位置变化（P页 / S选槽 / 品位）</div><div class=\"summary\">")
                .Append("本页 ").Append(pageNotes.Length).Append(" 个音符　·　全曲风险 ").Append(chart.AtRiskNoteCount)
                .Append("　不可演奏 ").Append(chart.UnplayableNoteCount);
            if (pageNotes.Length == 0) html.Append("　<span class=\"warning\">本曲无可映射到游戏键 1–6 的音符</span>");
            html.Append("</div></section>");
        }

        return html.Append("</body></html>").ToString();
    }

    private static string E(string value) => WebUtility.HtmlEncode(value ?? string.Empty);
    private static int LowerBound(ChartNote[] notes, long timeUs)
    {
        int low = 0, high = notes.Length;
        while (low < high)
        {
            int mid = low + ((high - low) / 2);
            if (notes[mid].StartUs < timeUs) low = mid + 1;
            else high = mid;
        }
        return low;
    }

    {

        return $"{(long)time.TotalMinutes:00}:{time.Seconds:00}.{time.Milliseconds:000}";
    }
    private static string FormatDuration(long us) => us <= 0 ? "" : us < 1_000_000
        ? $"{Math.Max(1, (int)Math.Round(us / 1000d))} ms"
        : $"{(us / 1_000_000d).ToString("0.#", CultureInfo.InvariantCulture)} s";
```

```csharp
public interface IPositionTimingModel
{

}

/// <summary>
/// Conservative starter model. Keeps linear distance for benchmark / basic testing.
/// </summary>
public sealed class LinearPositionTimingModel : IPositionTimingModel
{
    public long EstimatePrepareUs(GuitarPosition from, GuitarPosition to, ChartCompileOptions options)
    {
        var pageSteps = Math.Abs(to.Page - from.Page);
        var pageUs = pageSteps * (options.PageKeyHoldUs + options.PageInterTapGapUs);
        var slotUs = from.Slot == to.Slot ? 0 : options.WheelMouseSettleUs;
        return pageUs + slotUs + options.SafetyMarginUs;
    }
}

/// <summary>
/// Production calibrated timing model connected to the real WheelController and game navigation graph.
/// Accurately handles:
/// 1) P5 (open string 0-fret) after P4 in normal page order;
/// 2) Slot invalidation upon wheel page changes;
/// 3) Configurable timing parameters (KeyHold, InterTapGap, WheelMouseSettle, SafetyMargin).
/// </summary>
public sealed class WheelPositionTimingModel : IPositionTimingModel
{
    private readonly Func<ChartCompileOptions, (long HoldUs, long GapUs, long SettleUs, long SafetyUs)>? _paramOverrides;

    public WheelPositionTimingModel(Func<ChartCompileOptions, (long HoldUs, long GapUs, long SettleUs, long SafetyUs)>? paramOverrides = null)
    {
        _paramOverrides = paramOverrides;
    }

    public long EstimatePrepareUs(GuitarPosition from, GuitarPosition to, ChartCompileOptions options)
    {
        long holdUs = options.PageKeyHoldUs;
        long gapUs = options.PageInterTapGapUs;
        long settleUs = options.WheelMouseSettleUs;
        long safetyUs = options.SafetyMarginUs;

        if (_paramOverrides != null)
        {
            var (oHold, oGap, oSettle, oSafety) = _paramOverrides(options);
            if (oHold > 0) holdUs = oHold;
            if (oGap > 0) gapUs = oGap;
            if (oSettle > 0) settleUs = oSettle;
            if (oSafety > 0) safetyUs = oSafety;
        }

        // 1. Same position entirely: no movement or click required
        if (from.Page == to.Page && from.Slot == to.Slot)
        {
            return 0;
        }

        // 2. Intra-page slot change (no page flip)
        if (from.Page == to.Page)
        {
            // For P5, fret 0 is uniquely slot 1.
            if (to.Page == 5) return 0;
            return settleUs + safetyUs;
        }

        // 3. Inter-page navigation: P1..P5 are consecutive pages.
        int pageSteps = Math.Abs(to.Page - from.Page);

        long pageNavUs = pageSteps * (holdUs + gapUs);

        // When a page flips, WheelController.InvalidateCurrentSlot() is triggered.
        // The slot on the newly opened page must be clicked and settled.
        long slotSelectUs = settleUs;

        return pageNavUs + slotSelectUs + safetyUs;
    }
}
```

```csharp
using System.Diagnostics;
using System.Runtime.InteropServices;
using System.Collections.Concurrent;
using Nte.GuitarChart.Core;

namespace MidiAutoPlayer.SemiAuto.Input;

public sealed class NativeInputObserver : IUserInputObserver, IDisposable
{
    private const int WH_KEYBOARD_LL = 13;
    private const int WM_KEYDOWN = 0x0100;
    private const int WM_KEYUP = 0x0101;
    private const int WM_SYSKEYDOWN = 0x0104;
    private const int WM_SYSKEYUP = 0x0105;
    private const uint LLKHF_INJECTED = 0x00000010;

    [StructLayout(LayoutKind.Sequential)]
    private struct KBDLLHOOKSTRUCT
    {
        public uint vkCode;
        public uint scanCode;
        public uint flags;
        public uint time;
        public IntPtr dwExtraInfo;
    }

    private delegate IntPtr LowLevelKeyboardProc(int nCode, IntPtr wParam, IntPtr lParam);

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern IntPtr SetWindowsHookEx(int idHook, LowLevelKeyboardProc lpfn, IntPtr hMod, uint dwThreadId);

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool UnhookWindowsHookEx(IntPtr hhk);

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern IntPtr CallNextHookEx(IntPtr hhk, int nCode, IntPtr wParam, IntPtr lParam);

    [DllImport("kernel32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern IntPtr GetModuleHandle(string? lpModuleName);

    public event EventHandler<InputObservation>? InputObserved;
    public event Action<long>? PositionSwitchKeyPressed;
    public event Action<int, int, bool>? KeyStateChanged;
    public event Action<int, int, int>? LaneKeyCaptured;

    private IntPtr _hookId = IntPtr.Zero;
    private readonly LowLevelKeyboardProc _proc;
    private readonly ConcurrentDictionary<int, byte> _downKeys = new();
    private readonly ConcurrentDictionary<int, byte> _captureSuppressedKeys = new();
    private readonly ConcurrentDictionary<int, byte> _immersionCaptureKeys = new();
    private readonly BlockingCollection<Action> _dispatchQueue = new(new ConcurrentQueue<Action>());
    private readonly Thread _dispatchThread;
    private PhysicalKeyBinding[] _gameLaneBindings = GameLaneKeyBinding.CreateDefaultGameLanes().ToArray();
    private int _captureLane;
    private int _captureImmersionHotkey;

    public IReadOnlyList<PhysicalKeyBinding> GameLaneBindings => Volatile.Read(ref _gameLaneBindings);
    public event Action<int, int, bool>? ImmersionHotkeyCaptured;

    public void BeginImmersionHotkeyCapture() => Interlocked.Exchange(ref _captureImmersionHotkey, 1);
    public void CancelImmersionHotkeyCapture() => Interlocked.Exchange(ref _captureImmersionHotkey, 0);

    public void SetGameLaneBindings(IReadOnlyList<PhysicalKeyBinding> bindings)
    {
        if (bindings.Count != 6 || !GameLaneKeyBinding.AreUnique(bindings))
            throw new ArgumentException("Exactly six unique game lane key bindings are required.", nameof(bindings));
        Volatile.Write(ref _gameLaneBindings, bindings.Select(k => new PhysicalKeyBinding(k.VirtualKey, k.ScanCode)).ToArray());
    }

    public bool BeginLaneKeyCapture(int lane)
    {
        if (lane is < 1 or > 7) return false;
        Volatile.Write(ref _captureLane, lane);
        return true;
    }

    public void CancelLaneKeyCapture() => Volatile.Write(ref _captureLane, 0);

    public int PositionSwitchVk { get; set; } = PositionSwitchKeyBinding.DefaultVirtualKey;
    public int PositionSwitchScanCode { get; set; }
    public bool MatchesPositionSwitchKey(int virtualKey, int scanCode) =>
        PositionSwitchScanCode > 0 ? PositionSwitchScanCode == scanCode : PositionSwitchVk == virtualKey;

    public NativeInputObserver()
    {
        _proc = HookCallback;
        _dispatchThread = new Thread(DispatchLoop) { IsBackground = true, Name = "NTE_InputDispatch" };
        _dispatchThread.Start();
    }

    public bool Start()
    {
        if (_hookId != IntPtr.Zero) return true;
        using var curProcess = Process.GetCurrentProcess();
        using var curModule = curProcess.MainModule;
        _hookId = SetWindowsHookEx(WH_KEYBOARD_LL, _proc, GetModuleHandle(curModule?.ModuleName), 0);
        return _hookId != IntPtr.Zero;
    }

    public void Stop()
    {
        if (_hookId != IntPtr.Zero)
        {
            UnhookWindowsHookEx(_hookId);
            _hookId = IntPtr.Zero;
        }
        _downKeys.Clear();
        _captureSuppressedKeys.Clear();
        _immersionCaptureKeys.Clear();
        CancelImmersionHotkeyCapture();
        CancelLaneKeyCapture();
    }

    private IntPtr HookCallback(int nCode, IntPtr wParam, IntPtr lParam)
    {
        bool suppressCapturedInput = false;
        if (nCode >= 0)
        {
            int msg = wParam.ToInt32();
            bool isDown = (msg == WM_KEYDOWN || msg == WM_SYSKEYDOWN);
            bool isUp = (msg == WM_KEYUP || msg == WM_SYSKEYUP);

            if (isDown || isUp)
            {
                var kb = Marshal.PtrToStructure<KBDLLHOOKSTRUCT>(lParam);
                // Ignore SendInput events generated by this app (or other injectors).
                // They must not enter the physical-input state/capture pipeline.
                if ((kb.flags & LLKHF_INJECTED) != 0)
                    return CallNextHookEx(_hookId, nCode, wParam, lParam);

                int vk = (int)kb.vkCode;
                long wallClockUs = Stopwatch.GetTimestamp() * 1_000_000L / Stopwatch.Frequency;

                bool firstDown = isDown && _downKeys.TryAdd(-vk, 0);
                bool isModifier = IsModifierKey(vk);
                bool captureImmersionKey = firstDown && !isModifier
                    && Volatile.Read(ref _captureImmersionHotkey) != 0
                    && Interlocked.Exchange(ref _captureImmersionHotkey, 0) != 0;
                if (captureImmersionKey)
                {
                    bool hasModifiers = IsAnyModifierDown();
                    int scanCode = (int)kb.scanCode;
                    _immersionCaptureKeys.TryAdd(vk, 0);
                    _dispatchQueue.TryAdd(() => ImmersionHotkeyCaptured?.Invoke(vk, scanCode, hasModifiers));
                }

                bool isImmersionCaptureKey = _immersionCaptureKeys.ContainsKey(vk);
                int captureLane = Volatile.Read(ref _captureLane);
                if (!isImmersionCaptureKey && firstDown && captureLane is >= 1 and <= 7)
                {
                    _captureSuppressedKeys.TryAdd(vk, 0);
                    int scanCode = (int)kb.scanCode;
                    _dispatchQueue.TryAdd(() => LaneKeyCaptured?.Invoke(captureLane, vk, scanCode));
                }
                bool suppressForCapture = !isImmersionCaptureKey && _captureSuppressedKeys.ContainsKey(vk);
                if (isImmersionCaptureKey)
                {
                    if (isUp)
                    {
                        _immersionCaptureKeys.TryRemove(vk, out _);
                        _downKeys.TryRemove(-vk, out _);
                    }
                }
                else if (suppressForCapture)
                {
                    suppressCapturedInput = true;
                    if (isUp)
                    {
                        _captureSuppressedKeys.TryRemove(vk, out _);
                        _downKeys.TryRemove(-vk, out _);
                    }
                }
                else
                {
                    if (firstDown)
                    {
                        _dispatchQueue.TryAdd(() => KeyStateChanged?.Invoke(vk, (int)kb.scanCode, true));
                    }
                    else if (isUp)
                    {
                        _downKeys.TryRemove(-vk, out _);
                        _dispatchQueue.TryAdd(() => KeyStateChanged?.Invoke(vk, (int)kb.scanCode, false));
                    }

                    // Check position switch key
                    if (firstDown && MatchesPositionSwitchKey(vk, (int)kb.scanCode))
                    {
                        _dispatchQueue.TryAdd(() => PositionSwitchKeyPressed?.Invoke(wallClockUs));
                    }

                    int gameKey = GameLaneKeyBinding.MapPhysicalKeyToGameLane(vk, (int)kb.scanCode, GameLaneBindings);
                    if (gameKey >= 1 && gameKey <= 6)
                    {
                        bool isRepeat = isDown && !_downKeys.TryAdd(gameKey, 0);
                        if (isUp) _downKeys.TryRemove(gameKey, out _);
                        var observation = new InputObservation(gameKey, isDown, wallClockUs, isRepeat);
                        _dispatchQueue.TryAdd(() => InputObserved?.Invoke(this, observation));
                    }
                }
            }
        }

        if (suppressCapturedInput) return (IntPtr)1;
        return CallNextHookEx(_hookId, nCode, wParam, lParam);
    }

    private bool IsAnyModifierDown() =>
        _downKeys.ContainsKey(-0x10) || _downKeys.ContainsKey(-0x11) || _downKeys.ContainsKey(-0x12)
        || _downKeys.ContainsKey(-0x5B) || _downKeys.ContainsKey(-0x5C)
        || _downKeys.ContainsKey(-0xA0) || _downKeys.ContainsKey(-0xA1)
        || _downKeys.ContainsKey(-0xA2) || _downKeys.ContainsKey(-0xA3)
        || _downKeys.ContainsKey(-0xA4) || _downKeys.ContainsKey(-0xA5);

    private static bool IsModifierKey(int virtualKey) => virtualKey is
        0x10 or 0x11 or 0x12 or 0x5B or 0x5C or 0xA0 or 0xA1 or 0xA2 or 0xA3 or 0xA4 or 0xA5;

    private void DispatchLoop()
    {
        foreach (var action in _dispatchQueue.GetConsumingEnumerable())
        {
            try { action(); } catch { /* An observer callback must never terminate input dispatch. */ }
        }
    }

    public static int MapVirtualKeyToGameLane(int vk, int scanCode)
    {
        // 1..6 on top number row (0x31..0x36)
        if (vk >= 0x31 && vk <= 0x36)
            return vk - 0x30;

        // 1..6 on Numpad (0x61..0x66)
        if (vk >= 0x61 && vk <= 0x66)
            return vk - 0x60;

        return scanCode switch
        {
            0x4F => 1, 0x50 => 2, 0x51 => 3,
            0x4B => 4, 0x4C => 5, 0x4D => 6,
            _ => 0
        };
    }

    public static int MapConfiguredKeyToGameLane(int vk, int scanCode, IReadOnlyList<PhysicalKeyBinding> bindings) =>
        GameLaneKeyBinding.MapPhysicalKeyToGameLane(vk, scanCode, bindings);

    public void Dispose()
    {
        Stop();
        _dispatchQueue.CompleteAdding();
        if (_dispatchThread.IsAlive && _dispatchThread != Thread.CurrentThread) _dispatchThread.Join(500);
        _dispatchQueue.Dispose();
    }
}
```

```csharp
    public static IReadOnlyList<AutoShowcaseInputEvent> Create(

        foreach (var note in chart.Notes)
        {

                || note.Position is null || note.Status == FeasibilityStatus.Unplayable)
                continue;

            int stringIndex = note.PhysicalStringNo.Value - 1;
            ushort scanCode = configuredStringScanCodes is { Count: >= 6 }
                              && configuredStringScanCodes[stringIndex] != 0
                ? configuredStringScanCodes[stringIndex]
                : GameScanCodes.ForGuitarString(note.PhysicalStringNo.Value);
            long durationUs = Math.Clamp(note.DurationUs, 25_000L, 250_000L);
            events.Add(new(note.StartUs, scanCode, true, note.Id, note.GameKey.Value, note.Status));
            events.Add(new(note.StartUs + durationUs, scanCode, false, note.Id, note.GameKey.Value, note.Status));
        }

        return events.OrderBy(e => e.TimeUs).ThenBy(e => e.IsDown ? 1 : 0).ThenBy(e => e.GameKey).ToArray();
```

```csharp
        ScanCode = scanCode;

/// <summary>Defaults, validation and physical-key matching for the six playable lanes.</summary>
public static class GameLaneKeyBinding
{
    private static readonly PhysicalKeyBinding[] Defaults =
    [
        new(0x31, 0x02), new(0x32, 0x03), new(0x33, 0x04),
        new(0x34, 0x05), new(0x35, 0x06), new(0x36, 0x07)
    ];

    public static IReadOnlyList<PhysicalKeyBinding> CreateDefaultGameLanes() =>
        Defaults.Select(key => new PhysicalKeyBinding(key.VirtualKey, key.ScanCode)).ToArray();

    public static bool NormalizeConfig(MidiAutoPlayer.Models.AppConfig config)
    {
        if (config.SemiAutoGameLaneKeyBindings is null || config.SemiAutoGameLaneKeyBindings.Count != 6
            || config.SemiAutoGameLaneKeyBindings.Any(key => key is null))
        {
            config.SemiAutoGameLaneKeyBindings = CreateDefaultGameLanes().ToList();
            config.SemiAutoPositionSwitchVk = 0x4A;
            config.SemiAutoPositionSwitchScanCode = 0x24;
            return true;
        }

        var allKeys = config.SemiAutoGameLaneKeyBindings
            .Append(new PhysicalKeyBinding(config.SemiAutoPositionSwitchVk, config.SemiAutoPositionSwitchScanCode))
            .ToArray();
        bool gameKeysValid = config.SemiAutoGameLaneKeyBindings.All(k => IsAllowedPhysicalKey(k.VirtualKey, k.ScanCode, config.GlobalHotkeyVk));
        bool positionKeyValid = PositionSwitchKeyBinding.IsAllowedPhysicalKey(config.SemiAutoPositionSwitchVk, config.SemiAutoPositionSwitchScanCode)
            && config.SemiAutoPositionSwitchVk != config.GlobalHotkeyVk;
        if (!gameKeysValid || !positionKeyValid || !AreUnique(allKeys))
        {
            config.SemiAutoGameLaneKeyBindings = CreateDefaultGameLanes().ToList();
            config.SemiAutoPositionSwitchVk = 0x4A;
            config.SemiAutoPositionSwitchScanCode = 0x24;
            return true;
        }
        return false;
    }

    public static bool IsAllowedPhysicalKey(int virtualKey, int scanCode, int globalHotkeyVk = 0x77)
    {
        if (virtualKey is < 0x08 or > 0xFE || virtualKey == globalHotkeyVk) return false;
        if (virtualKey is 0x1B or 0x5B or 0x5C or 0x10 or 0x11 or 0x12 or >= 0xA0 and <= 0xA5) return false;
        if (virtualKey is 0x52 or 0x54 or 0x77 or 0x78 or 0x7B) return false;
        return true;
    }

    public static bool AreUnique(IReadOnlyList<PhysicalKeyBinding> bindings)
    {
        for (int i = 0; i < bindings.Count; i++)
        for (int j = i + 1; j < bindings.Count; j++)
        {
            var a = bindings[i];
            var b = bindings[j];
            if (a.ScanCode > 0 && b.ScanCode > 0)
            {
                if (a.ScanCode == b.ScanCode) return false;
            }
            else if (a.VirtualKey == b.VirtualKey)
            {
                return false;
            }
        }
        return true;
    }

    public static int MapPhysicalKeyToGameLane(int virtualKey, int scanCode, IReadOnlyList<PhysicalKeyBinding> bindings)
    {
        for (int i = 0; i < Math.Min(6, bindings.Count); i++)
        {
            var binding = bindings[i];
            if (scanCode > 0 && binding.ScanCode > 0 && scanCode == binding.ScanCode) return i + 1;
            if ((scanCode == 0 || binding.ScanCode == 0) && virtualKey != 0 && virtualKey == binding.VirtualKey) return i + 1;
        }
        return 0;
    }

    public static string FormatLabel(int virtualKey, int scanCode = 0)
    {
        return scanCode switch
        {
            0x4F => "Num 1", 0x50 => "Num 2", 0x51 => "Num 3",
            0x4B => "Num 4", 0x4C => "Num 5", 0x4D => "Num 6",
            _ when virtualKey is >= 0x30 and <= 0x39 => ((char)virtualKey).ToString(),
            _ when virtualKey is >= 0x41 and <= 0x5A => ((char)virtualKey).ToString(),
            _ when virtualKey is >= 0x70 and <= 0x87 => $"F{virtualKey - 0x6F}",
            0x01 => "鼠标左键", 0x02 => "鼠标右键", 0x04 => "鼠标中键",
            _ => $"VK 0x{virtualKey:X2}"
        };
    }
}
```

```csharp
/// <summary>Monotonic MIDI playhead advanced only while the practice driver key is held.</summary>
internal sealed class PracticeTimeline
{
    private long _chartOffsetUs;
    private long _wallAnchorUs;
    private double _speedRatio = 1.0;

    public bool IsRunning { get; private set; }

    public void Start(long chartOffsetUs, long wallClockUs, int speedPercent)
    {
        _chartOffsetUs = Math.Max(0, chartOffsetUs);
        _wallAnchorUs = wallClockUs;
        _speedRatio = NormalizeSpeed(speedPercent);
        IsRunning = true;
    }

    public void Resume(long wallClockUs)
    {
        if (IsRunning) return;
        _wallAnchorUs = wallClockUs;
        IsRunning = true;
    }

    public void Pause(long wallClockUs)
    {
        if (!IsRunning) return;
        _chartOffsetUs = CurrentTimeUs(wallClockUs);
        IsRunning = false;
    }

    public long CurrentTimeUs(long wallClockUs) => IsRunning
        ? Math.Max(0, _chartOffsetUs + (long)Math.Max(0, (wallClockUs - _wallAnchorUs) * _speedRatio))
        : _chartOffsetUs;

    public void Seek(long chartTimeUs, long wallClockUs)
    {
        _chartOffsetUs = Math.Max(0, chartTimeUs);
        _wallAnchorUs = wallClockUs;
    }

    public void SetSpeed(int speedPercent, long wallClockUs)
    {
        _chartOffsetUs = CurrentTimeUs(wallClockUs);
        _wallAnchorUs = wallClockUs;
        _speedRatio = NormalizeSpeed(speedPercent);
    }

    private static double NormalizeSpeed(int speedPercent) => Math.Clamp(speedPercent, 25, 200) / 100d;
```

```csharp
    private void ApplyLayoutBounds(Rectangle workArea, bool alignRight)
    {
        int availableWidth = Math.Max(1, workArea.Width - 24);
        int availableHeight = Math.Max(1, workArea.Height - 24);
        double dpiScale = Math.Max(1d, DeviceDpi / 96d);
        int width, height;
        if (_wideLayout)
        {
            int minWidth = Math.Min((int)Math.Round(760 * dpiScale), availableWidth);
            int maxWidth = Math.Min((int)Math.Round(1400 * dpiScale), availableWidth);
            width = Math.Clamp((int)Math.Round(workArea.Width * .72), minWidth, Math.Max(minWidth, maxWidth));
            int minHeight = Math.Min((int)Math.Round(520 * dpiScale), availableHeight);
            int maxHeight = Math.Min((int)Math.Round(850 * dpiScale), availableHeight);
            height = Math.Clamp((int)Math.Round(workArea.Height * .66), minHeight, Math.Max(minHeight, maxHeight));
            MinimumSize = new Size(width, Math.Min(height, minHeight));
            MaximumSize = new Size(width, availableHeight);
        }
        else
        {
            height = Math.Min(availableHeight, Math.Clamp((int)Math.Round(workArea.Height * .86), (int)Math.Round(620 * dpiScale), (int)Math.Round(1360 * dpiScale)));
            int minWidth = Math.Min((int)Math.Round(380 * dpiScale), availableWidth);
            int maxWidth = Math.Min((int)Math.Round(500 * dpiScale), availableWidth);
            width = Math.Clamp((int)Math.Round(height * .52), minWidth, Math.Max(minWidth, maxWidth));
            // Keep the portrait layout narrow while allowing its existing vertical sizing behavior.
            MinimumSize = new Size(width, Math.Min(560, availableHeight));
            MaximumSize = new Size(width, availableHeight);
        }
        int x = alignRight ? workArea.Right - width - 12 : Math.Clamp(Left, workArea.Left, Math.Max(workArea.Left, workArea.Right - width));
        int y = alignRight
            ? (_wideLayout ? workArea.Top + 12 : workArea.Top + (workArea.Height - height) / 2)
            : Math.Clamp(Top, workArea.Top, Math.Max(workArea.Top, workArea.Bottom - height));
        Bounds = new Rectangle(x, y, width, height);
    }

    private void ConstrainToCurrentScreen(Screen? targetScreen = null)
    {
        if (_constrainingBounds || !IsHandleCreated) return;
        // During a cross-monitor native drag, the release point is the clearest
        // indication of the user's intended destination (a straddling window
        // can otherwise be assigned to the monitor with the largest overlap).
        var screen = targetScreen ?? Screen.FromRectangle(Bounds);
        if (_activeScreenDeviceName != screen.DeviceName)
        {
            _constrainingBounds = true;
            try { ApplyLayoutBounds(screen, alignRight: false); }
            finally { _constrainingBounds = false; }
            return;
        }
        var work = screen.WorkingArea;
        int width = Math.Min(Width, work.Width), height = Math.Min(Height, work.Height);
        int x = Math.Clamp(Left, work.Left, Math.Max(work.Left, work.Right - width));
        int y = Math.Clamp(Top, work.Top, Math.Max(work.Top, work.Bottom - height));
        if (x == Left && y == Top && width == Width && height == Height) return;
        _constrainingBounds = true;
        try { Bounds = new Rectangle(x, y, width, height); }
        finally { _constrainingBounds = false; }
    }

    protected override void OnDpiChanged(DpiChangedEventArgs e)
    {
        base.OnDpiChanged(e);
        var screen = Screen.FromRectangle(Bounds);
        _constrainingBounds = true;
        try { ApplyLayoutBounds(screen, alignRight: false); }
        finally { _constrainingBounds = false; }
    }
```

```csharp
using System.Collections.Concurrent;
using Nte.GuitarChart.Core;

namespace Nte.GuitarChart.WinForms;
using GuitarChartModel = Nte.GuitarChart.Core.GuitarChart;

/// <summary>Time-aligned practice targets. Judging a note never edits or reorders the source chart.</summary>
internal sealed class PracticeSessionController
{
    public const long SimultaneousToleranceUs = 20_000;
    private readonly IReadOnlyList<ChartNote> _targets;
    private readonly ConcurrentDictionary<Guid, byte> _judged = new();

    public PracticeEngine Engine { get; } = new();
    public int JudgedCount => _judged.Count;
    public int TotalTargets => _targets.Count;
    public bool IsCompleted => _judged.Count >= _targets.Count;
    public ChartNote? Current => GetUpcomingTargets(0).FirstOrDefault();
    public IReadOnlyList<ChartNote> Upcoming => GetUpcomingTargets(Current?.StartUs ?? 0).Skip(1).Take(3).ToArray();

    public PracticeSessionController(GuitarChartModel chart, int startIndex = 0)
    {
        _targets = chart.Notes
            .Where(n => n.GameKey is >= 1 and <= 6 && n.Position.HasValue && n.Status != FeasibilityStatus.Unplayable)
            .OrderBy(n => n.StartUs)
            .ThenBy(n => n.GameKey)
            .ToArray();

        if (startIndex > 0)
        {
            foreach (var note in _targets.Take(Math.Clamp(startIndex, 0, _targets.Count)))
                _judged.TryAdd(note.Id, 0);
        }
    }

    public IReadOnlyList<ChartNote> GetTargetsAt(long timeUs, long toleranceUs = 80_000) =>
        _targets.Where(n => !_judged.ContainsKey(n.Id)
                && Math.Abs(n.StartUs - timeUs) <= Math.Max(0, toleranceUs))
            .OrderBy(n => Math.Abs(n.StartUs - timeUs))
            .ThenBy(n => n.GameKey)
            .ToArray();

    public IReadOnlyList<ChartNote> GetUpcomingTargets(long timeUs)
    {
        var next = _targets.FirstOrDefault(n => !_judged.ContainsKey(n.Id) && n.StartUs >= timeUs);
        if (next == null) return Array.Empty<ChartNote>();
        return _targets.Where(n => !_judged.ContainsKey(n.Id)
                && Math.Abs(n.StartUs - next.StartUs) <= SimultaneousToleranceUs)
            .OrderBy(n => n.GameKey)
            .ToArray();
    }

    public long? GetTailRecoveryTimeUs(long currentTimeUs, long durationUs)
    {
        if (currentTimeUs < durationUs) return null;
        return _targets.FirstOrDefault(note => !_judged.ContainsKey(note.Id))?.StartUs;
    }

    public PracticeFeedback EvaluateInput(int gameKey, long actualTimeUs, long toleranceUs = 80_000)
    {
        var active = GetTargetsAt(actualTimeUs, toleranceUs);
        if (active.Count == 0)
            return new PracticeFeedback("early", "当前时间附近没有待练音符。");

        var match = active.FirstOrDefault(n => n.GameKey == gameKey);
        var expected = match ?? active[0];
        var feedback = Engine.EvaluateKey(expected.GameKey!.Value, gameKey, expected.Id);
        if (feedback.Kind == "correct") _judged.TryAdd(expected.Id, 0);
        return feedback;
    }
}
```

```csharp
            _currentTimeUs = _practiceSession.Current?.StartUs ?? 0;
        }
        var target = _practiceSession.GetUpcomingTargets(_currentTimeUs).FirstOrDefault() ?? _practiceSession.Current;
        if (target == null)
        {
            PublishPracticeState("本段完成");
            PublishPlaybackTick("本段完成", false);
            return Task.CompletedTask;
        }
        if (!skipInitialPosition && target?.Position is { } position && !PositionTo(position, initialPage)) return Task.CompletedTask;
        _window.SetClickThrough(false);
        _clickThroughChanged?.Invoke(false);
        if (!_inputObserver.Start()) { Toast("error", $"全局键盘监听启动失败 (Win32 {_lastError()})，无法开始练习。"); return Task.CompletedTask; }
        _isPlaying = true;
        _stopRequested = false;
        _practiceDriverKey = 0;
        long nowUs = MonotonicTimeUs();
        _practiceTimeline.Pause(nowUs);
        _practiceTimeline.Seek(_currentTimeUs, nowUs);
        PublishPracticeState("等待按下当前目标键");
        PublishPlaybackTick("练习已就绪，等待按键", false);
        _playbackThread = new Thread(PracticeTimelineLoop)
        {
            IsBackground = true,
            Priority = ThreadPriority.AboveNormal,
            Name = "NTE_PracticeTimeline"
        };
        _playbackThread.Start();
        StartAutomaticImmersion();
        return Task.CompletedTask;
    }

    private async Task StartRhythm(int? initialPage, bool skipInitialPosition = false)
    {
        if (_currentMode == PlayMode.Simulated && !_simulatedMidiPlayer.IsAvailable)
            Toast("warning", "系统 MIDI 输出设备不可用：谱面和按键判定仍会继续，但无法输出声音。");
        if (_currentMode == PlayMode.AutoShowcase && (_config == null || _config.TargetWindowHandle == IntPtr.Zero || !KeyboardSender.IsForeground(_config.TargetWindowHandle)))
        {
            Toast("warning", "锁屏可打需要先绑定游戏窗口，并将游戏切到前台。自动演奏会向该窗口发送按键。");
            return;
        }
        if (_currentMode == PlayMode.AutoShowcase && _currentChart!.UnplayableNoteCount > 0)
            Toast("warning", $"当前曲谱有 {_currentChart.UnplayableNoteCount} 个音符无法共用同一轮盘位置，锁屏可打会跳过这些音符。");
        if (_rhythmCountdown != null) return;
        bool resume = _currentTimeUs > 0 && _currentTimeUs < _currentChart!.DurationUs;
        if (!resume) { _currentTimeUs = 0; _judgedNotes.Clear(); _judgedSwitches.Clear(); ResetEngines(); }
        if (!skipInitialPosition)
        {
            if (_currentMode == PlayMode.Simulated)
            {
                // MIDI follow mode plays the complete chart without controlling game position.
            }
            else if (_executionMode == PositionExecutionMode.SemiAuto)
            {
                var position = _currentChart!.Notes.FirstOrDefault(n => n.StartUs >= _currentTimeUs && n.Position.HasValue)?.Position;
                if (position.HasValue && !PositionTo(position.Value, initialPage)) return;
            }
            else
            {
                var position = _currentChart!.Notes.FirstOrDefault(n => n.StartUs >= _currentTimeUs && n.Position.HasValue)?.Position;
                if (position.HasValue && !PositionTo(position.Value, _currentMode == PlayMode.AutoShowcase ? initialPage : null) && _currentMode == PlayMode.AutoShowcase) return;
            }
        }
        _window.SetClickThrough(false);
        _clickThroughChanged?.Invoke(false);
        if (_currentMode is PlayMode.RhythmNormal or PlayMode.RhythmHell or PlayMode.Simulated)
        {
            var countdown = new CancellationTokenSource();
            _rhythmCountdown = countdown;
            try
            {
                for (int seconds = 3; seconds > 0; seconds--)
                {
                    _bridge?.Publish(new { type = "playback.countdown", payload = new { active = true, seconds } });
                    await Task.Delay(1000, countdown.Token);
                }
            }
            catch (OperationCanceledException) when (countdown.IsCancellationRequested)
            {
                return;
            }
            finally
            {
                bool ownsCountdown = ReferenceEquals(_rhythmCountdown, countdown);
                if (ownsCountdown) _rhythmCountdown = null;
                countdown.Dispose();
                if (ownsCountdown)
                    _bridge?.Publish(new { type = "playback.countdown", payload = new { active = false, seconds = 0 } });
            }
        }
        _isPlaying = true;
        _stopRequested = false;
        _pauseEvent.Set();
        _rhythmSession.Start(_currentTimeUs);
        if (_currentMode != PlayMode.AutoShowcase && !_inputObserver.Start())
        {
            _isPlaying = false;
            _rhythmSession.Stop();
            Toast("error", $"全局键盘监听启动失败 (Win32 {_lastError()})，无法开始音游。");
            return;
        }
        if (_currentMode == PlayMode.Simulated) StartSimulatedMidiPlayback(_currentTimeUs);
        if (_currentMode == PlayMode.AutoShowcase)
        {
            _autoInputThread = new Thread(AutoShowcaseInputLoop)
            {
                IsBackground = true,
                Priority = ThreadPriority.Highest,
                Name = "NTE_AutoShowcaseInput"
            };
            _autoInputThread.Start();
        }
        _playbackThread = new Thread(RhythmPlaybackLoop) { IsBackground = true, Priority = ThreadPriority.AboveNormal, Name = "NTE_RhythmPlayback" };
        _playbackThread.Start();
        StartAutomaticImmersion(force: _currentMode == PlayMode.AutoShowcase);
    }

    private bool PositionTo(GuitarPosition position, int? initialPage = null)
    {
        if (_executionMode == PositionExecutionMode.FullManual && _currentMode != PlayMode.AutoShowcase)
        {
            _bridge?.Publish(new { type = "position.prompt", payload = new { targetPage = position.Page, targetSlot = position.Slot, fret = position.Fret, stateText = $"请手动切换至 P{position.Page} · Slot {position.Slot} · {position.Fret}品", cueTimeUs = _currentTimeUs } });
            return true;
        }
        if (_autoPosition == null) return true;
        try
        {
            if (_autoPosition(position, initialPage))
            {
                _bridge?.Publish(new { type = "position.active", payload = new { page = position.Page, slot = position.Slot, stateText = $"已定位至 P{position.Page} · Slot {position.Slot}" } });
                return true;
            }
            Toast("warning", $"无法定位到 P{position.Page} · Slot {position.Slot}，请绑定游戏窗口并确认游戏处于前台。");
            return false;
        }
        catch (Exception ex) { Toast("error", $"位置切换失败：{ex.Message}"); return false; }
    }

    public Task PauseAsync()

        {

    public Task SeekAsync(double ratio)
    {
        var chart = _currentChart;
        if (chart == null || chart.DurationUs <= 0) return Task.CompletedTask;

        ratio = Math.Clamp(ratio, 0d, 1d);
        bool resume = _isPlaying;
        StopPlayback();
        _judgedNotes.Clear();
        _judgedSwitches.Clear();

        if (_currentMode == PlayMode.Practice)
        {
            _practiceSession = new PracticeSessionController(chart);
            _currentTimeUs = (long)(chart.DurationUs * ratio);
            _practiceTimeline.Seek(_currentTimeUs, MonotonicTimeUs());
            var practicePosition = _practiceSession.GetUpcomingTargets(_currentTimeUs).FirstOrDefault()?.Position;
            if (!resume && practicePosition is { } targetPosition) PositionTo(targetPosition);
            PublishPracticeState(_practiceSession.GetUpcomingTargets(_currentTimeUs).Count == 0 ? "本段完成" : "已定位，按开始后继续练习");
        }
        else
        {
            _currentTimeUs = (long)(chart.DurationUs * ratio);
            ResetEngines();
            if (_executionMode == PositionExecutionMode.SemiAuto)
            {
                var position = chart.Notes.FirstOrDefault(n => n.StartUs >= _currentTimeUs && n.Position.HasValue)?.Position;
                if (position.HasValue) PositionTo(position.Value);
            }
            else
            {
                var position = chart.Notes.FirstOrDefault(n => n.StartUs >= _currentTimeUs && n.Position.HasValue)?.Position;
                if (position.HasValue) PositionTo(position.Value);
            }
        }

        PublishPlaybackTick("已定位", false);
        if (resume)
        {
            if (_currentMode == PlayMode.Practice)
            {
                var startTask = StartPractice(initialPage: null, skipInitialPosition: true);
                if (_practiceSession?.GetUpcomingTargets(_currentTimeUs).FirstOrDefault()?.Position is { } practicePosition)
                    QueuePositionTo(practicePosition);
                return startTask;
            }
            return StartRhythm(initialPage: null, skipInitialPosition: true);
        }
        return Task.CompletedTask;
    }

    public Task SetSpeedAsync(int speedPercent)
    {
        long nowUs = MonotonicTimeUs();
        if (_isPlaying)
            _currentTimeUs = _currentMode == PlayMode.Practice
                ? _practiceTimeline.CurrentTimeUs(nowUs)
                : _rhythmSession.CurrentTimeUs(_speedPercent);
        _speedPercent = Math.Clamp(speedPercent, 25, 200);
        _speedManager.SetUserManualSpeed(_speedPercent);
        _rhythmSession.SetSpeedRatio(_speedPercent / 100d);
        if (_currentMode == PlayMode.Practice) _practiceTimeline.SetSpeed(_speedPercent, nowUs);
        if (_isPlaying && _currentMode != PlayMode.Practice)
        {
            _rhythmSession.Start(_currentTimeUs);
            if (_currentMode == PlayMode.Simulated) StartSimulatedMidiPlayback(_currentTimeUs);
        }
        return Task.CompletedTask;
    }

    public Task BeginLaneKeyCaptureAsync(int lane)
    {
        if (lane is < 1 or > 7) throw new ArgumentOutOfRangeException(nameof(lane), "轨道编号必须为 1～7。");
        if (_isPlaying || _rhythmCountdown != null)
        {
            _bridge?.Publish(new { type = "lane-key.error", payload = new { message = "请先暂停演奏，再设置键位。" } });
            return Task.CompletedTask;

            _bridge?.Publish(new { type = "position-key.capture", payload = new { active = true, virtualKey = _inputObserver.PositionSwitchVk, label = FormatVirtualKey(_inputObserver.PositionSwitchVk, _inputObserver.PositionSwitchScanCode) } });

    private void OnImmersionHotkeyCaptured(int virtualKey, int scanCode, bool hasModifiers)
    {
        if (Volatile.Read(ref _immersionHotkeyCaptureActive) == 0) return;
        _ = CompleteImmersionHotkeyCaptureAsync(virtualKey, hasModifiers);
    }

    private async Task CompleteImmersionHotkeyCaptureAsync(int virtualKey, bool hasModifiers)
    {
        if (virtualKey == 0x1B)
        {
            await CancelImmersionHotkeyCaptureAsync();
            return;
        }

        if (hasModifiers)
        {
            if (Volatile.Read(ref _immersionHotkeyCaptureActive) == 0) return;
            _inputObserver.BeginImmersionHotkeyCapture();
            _bridge?.Publish(new { type = "immersion.hotkey.capture.state", payload = new { active = true, message = "只支持单个按键，不支持组合键；请重试，Esc 取消", error = true } });
            return;
        }

        if (Interlocked.Exchange(ref _immersionHotkeyCaptureActive, 0) == 0) return;
        _bridge?.Publish(new { type = "immersion.hotkey.capture.state", payload = new { active = false } });
        try
        {
            await SetImmersionHotkeyAsync(virtualKey);
        }
        catch (Exception ex)
        {
            _bridge?.Publish(new { type = "immersion.hotkey.error", payload = new { message = $"快捷键设置失败：{ex.Message}" } });
        }
        finally
        {
            if (!_isPlaying && _capturingLaneKey == 0) _inputObserver.Stop();
        }
    }

    public Task SetImmersionHotkeyAsync(int virtualKey)
    {
        if (virtualKey == (_config?.SemiAutoImmersionHotkeyVk ?? _window.ImmersionHotkeyVk))
        {
            PublishImmersionState();
            return Task.CompletedTask;
        }
        if (!IsAllowedImmersionHotkey(virtualKey))
        {
            _bridge?.Publish(new { type = "immersion.hotkey.error", payload = new { message = "该键与全局快捷键、轨道键或换位键冲突，或该键被系统保留。" } });
            return Task.CompletedTask;
        }
        if (!_window.TrySetImmersionHotkey(virtualKey))
        {
            _bridge?.Publish(new { type = "immersion.hotkey.error", payload = new { message = "全局按键监听启动失败，快捷键未更改。" } });
            return Task.CompletedTask;
        }
        _immersionHotkeyRegistrationFailed = false;
        if (_config != null)
        {
            _config.SemiAutoImmersionHotkeyVk = virtualKey;
            _saveConfig?.Invoke();
        }
        PublishImmersionState();
        return Task.CompletedTask;
    }

    private bool IsAllowedImmersionHotkey(int virtualKey) => ImmersionHotkeyBinding.IsAllowed(
        virtualKey,
        _config?.GlobalHotkeyVk ?? 0x77,
        _inputObserver.GameLaneBindings,
        new PhysicalKeyBinding(_inputObserver.PositionSwitchVk, _inputObserver.PositionSwitchScanCode));

    private int FindAllowedImmersionHotkey()
    {
        for (int key = ImmersionHotkeyBinding.DefaultVirtualKey; key <= 0x5A; key++)
            if (IsAllowedImmersionHotkey(key)) return key;
        for (int key = 0x41; key < ImmersionHotkeyBinding.DefaultVirtualKey; key++)
            if (IsAllowedImmersionHotkey(key)) return key;
        return ImmersionHotkeyBinding.DefaultVirtualKey;
    }

    private void OnLaneKeyCaptured(int lane, int virtualKey, int scanCode)
    {
        if (_capturingLaneKey != lane) return;
        if (virtualKey == 0x1B)
        {
            _capturingLaneKey = 0;
            _inputObserver.CancelLaneKeyCapture();
            if (!_isPlaying) _inputObserver.Stop();
            _bridge?.Publish(new { type = "lane-key.capture.state", payload = new { active = false, lane, cancelled = true } });
            return;
        }

        if (!GameLaneKeyBinding.IsAllowedPhysicalKey(virtualKey, scanCode, _config?.GlobalHotkeyVk ?? 0x77)
            || virtualKey == (_config?.SemiAutoImmersionHotkeyVk ?? ImmersionHotkeyBinding.DefaultVirtualKey)
            || (lane == 7 && !IsAllowedPositionKey(virtualKey, scanCode)))
        {
            _bridge?.Publish(new { type = "lane-key.error", payload = new { message = "该键为系统、快捷键或轨道控制保留键，请按其他键。" } });
            return;
        }

        var current = _inputObserver.GameLaneBindings.Select(k => new PhysicalKeyBinding(k.VirtualKey, k.ScanCode)).ToList();
        var position = new PhysicalKeyBinding(_inputObserver.PositionSwitchVk, _inputObserver.PositionSwitchScanCode);
        var candidate = new PhysicalKeyBinding(virtualKey, scanCode);
        if (lane == 7) position = candidate;
        else current[lane - 1] = candidate;
        if (!GameLaneKeyBinding.AreUnique(current.Append(position).ToArray()))
        {
            _bridge?.Publish(new { type = "lane-key.error", payload = new { message = "这个键已分配给其他轨道，请选择不同按键。" } });
            return;
        }

        _capturingLaneKey = 0;
        _inputObserver.CancelLaneKeyCapture();
        if (!_isPlaying) _inputObserver.Stop();
        if (lane == 7)
        {
            SetPositionSwitchKey(virtualKey, scanCode);
        }
        else
        {
            _inputObserver.SetGameLaneBindings(current);
            if (_config != null)
            {
                _config.SemiAutoGameLaneKeyBindings = current;
                _saveConfig?.Invoke();
            }
            PublishLaneKeyState();
            Toast("success", $"第 {lane} 轨已设置为 {GameLaneKeyBinding.FormatLabel(virtualKey, scanCode)}");
        }
        _bridge?.Publish(new { type = "lane-key.capture.state", payload = new { active = false, lane } });
    }

    private void SetPositionSwitchKey(int virtualKey, int scanCode)
    {
        var allKeys = _inputObserver.GameLaneBindings
            .Append(new PhysicalKeyBinding(virtualKey, scanCode)).ToArray();
        if (!IsAllowedPositionKey(virtualKey, scanCode)
            || virtualKey == (_config?.SemiAutoImmersionHotkeyVk ?? ImmersionHotkeyBinding.DefaultVirtualKey)
            || !GameLaneKeyBinding.AreUnique(allKeys))
        {

    private void RhythmPlaybackLoop()
    {
        var chart = _currentChart;
        if (chart == null) return;
        long durationUs = chart.DurationUs;
        long startOffsetUs = _currentTimeUs;
        int noteIndex = 0;
        while (noteIndex < chart.Notes.Count && chart.Notes[noteIndex].StartUs < startOffsetUs) noteIndex++;
        int eventIndex = 0;
        while (eventIndex < chart.PositionEvents.Count && chart.PositionEvents[eventIndex].TargetTimeUs < startOffsetUs) eventIndex++;
        int switchMissIndex = 0;
        while (switchMissIndex < chart.PositionEvents.Count && chart.PositionEvents[switchMissIndex].DeadlineTimeUs < startOffsetUs) switchMissIndex++;
        while (!_stopRequested && _currentTimeUs < durationUs + 500_000)
        {
            _pauseEvent.Wait();
            if (_stopRequested) break;
            _currentTimeUs = _rhythmSession.CurrentTimeUs(_speedPercent);
            if (_stopRequested) break;
            while (eventIndex < chart.PositionEvents.Count && chart.PositionEvents[eventIndex].PromptTimeUs <= _currentTimeUs)
            {
                var ev = chart.PositionEvents[eventIndex++];
                if (_currentMode != PlayMode.Simulated)
                {
                    if (_executionMode == PositionExecutionMode.SemiAuto || _currentMode == PlayMode.AutoShowcase)
                        QueuePositionTo(ev.To);
                    else
                        PositionTo(ev.To);
                    _bridge?.Publish(new { type = "position.prompt", payload = new { eventId = ev.Id, targetPage = ev.To.Page, targetSlot = ev.To.Slot, deadlineUs = ev.DeadlineTimeUs } });
                }
            }
            while (switchMissIndex < chart.PositionEvents.Count && chart.PositionEvents[switchMissIndex].DeadlineTimeUs < _currentTimeUs)
            {
                var expiredSwitch = chart.PositionEvents[switchMissIndex++];
                if (_executionMode != PositionExecutionMode.FullManual || _currentMode != PlayMode.RhythmHell || !_judgedSwitches.TryAdd(expiredSwitch.Id, 0)) continue;
            }
            long missWindowUs = 160_000L * _speedPercent / 100;
            while (_currentMode != PlayMode.AutoShowcase && noteIndex < chart.Notes.Count && chart.Notes[noteIndex].StartUs + missWindowUs < _currentTimeUs)
            {
                var expired = chart.Notes[noteIndex++];
                if (!_judgedNotes.TryAdd(expired.Id, 0)) continue;
                PublishNoteFeedback(expired, HitGrade.Miss);
            }
            while (noteIndex < chart.Notes.Count && chart.Notes[noteIndex].StartUs + chart.Notes[noteIndex].DurationUs <= _currentTimeUs) noteIndex++;
            PublishPlaybackTick("正在演奏", true);
            Thread.Sleep(16);
        }
        bool completed = !_stopRequested && _currentTimeUs >= durationUs;
        _isPlaying = false;
        _rhythmSession.Stop();
        EndAutomaticImmersion();
        _window.SetClickThrough(false);
        _clickThroughChanged?.Invoke(false);
        _inputObserver.Stop();
        ReleaseShowcaseKeys();
        if (completed) { _currentTimeUs = durationUs; PublishPlaybackTick("演奏已结束", false); }
    }

    private void QueuePositionTo(GuitarPosition position)
    {
        _positionQueue.Enqueue((position, Volatile.Read(ref _positionGeneration)));
        _positionSignal.Set();
    }

    private void PositionExecutionLoop()
    {
        while (!_positionThreadStop)
        {
            _positionSignal.WaitOne();
            if (_positionThreadStop) break;
            (GuitarPosition Position, int Generation)? latest = null;
            while (_positionQueue.TryDequeue(out var request)) latest = request;
            if (latest is not { } pending
                || pending.Generation != Volatile.Read(ref _positionGeneration)
                || !_isPlaying || _stopRequested)
                continue;
            PositionTo(pending.Position);
        }
    }

    private void AutoShowcaseInputLoop()
    {
        var chart = _currentChart;
        if (chart == null) return;
        var events = AutoShowcaseInputPlan.Create(chart, _config?.WheelPreset?.StringScanCodes, _inputObserver.GameLaneBindings);
        var notesById = chart.Notes.ToDictionary(note => note.Id);
        int index = 0;
        while (index < events.Count && events[index].TimeUs < _currentTimeUs) index++;
        long frequency = System.Diagnostics.Stopwatch.Frequency;
        while (!_stopRequested && index < events.Count)
        {
            _pauseEvent.Wait();
            if (_stopRequested) break;
            long nowUs = _rhythmSession.CurrentTimeUs(_speedPercent);
            long remainingUs = events[index].TimeUs - nowUs;
            if (remainingUs > 2_000)
            {
                Thread.Sleep((int)Math.Min(remainingUs / 1000 - 1, 10));
                continue;
            }
            if (remainingUs > 0)
            {
                long deadline = System.Diagnostics.Stopwatch.GetTimestamp() + remainingUs * frequency / 1_000_000;
                while (!_stopRequested && _pauseEvent.IsSet && System.Diagnostics.Stopwatch.GetTimestamp() < deadline)
                    Thread.SpinWait(32);
                continue;
            }

            var inputEvent = events[index++];
            if (inputEvent.IsDown)
            {
                bool alreadyHeld;
                lock (_autoHeldLock) alreadyHeld = _autoHeldScanCodes.Contains(inputEvent.ScanCode);
                if (alreadyHeld && !SendShowcaseKey(inputEvent with { IsDown = false })) { _stopRequested = true; break; }
            }
            if (!SendShowcaseKey(inputEvent)) { _stopRequested = true; break; }
            if (inputEvent.IsDown && _judgedNotes.TryAdd(inputEvent.NoteId, 0))
            {
                if (notesById.TryGetValue(inputEvent.NoteId, out var note))
                    PublishNoteFeedback(note, HitGrade.Perfect);
            }
        }
    }

    private void OnInputObserved(object? sender, InputObservation obs)
    {
        if (_currentMode == PlayMode.AutoShowcase) return;
        if (!obs.IsKeyDown)
        {
            _heldGameKeys.TryRemove(obs.GameKey, out _);
            if (_currentMode == PlayMode.Practice && _isPlaying && _practiceDriverKey == obs.GameKey)
            {
                _practiceTimeline.Pause(obs.WallClockUs);
                _currentTimeUs = _practiceTimeline.CurrentTimeUs(obs.WallClockUs);
                _practiceDriverKey = 0;
                EndAutomaticImmersion();
                PublishPracticeState("已松开，谱面暂停");
                PublishPlaybackTick("已松开，谱面暂停", false);
            }
            return;
        }
        if (!_isPlaying || obs.IsAutoRepeat) return;
        _heldGameKeys[obs.GameKey] = obs.WallClockUs;
        if (_currentMode == PlayMode.Practice) { HandlePracticeKey(obs); return; }
        HandleRhythmKey(obs);
    }

    private bool SendShowcaseKey(AutoShowcaseInputEvent inputEvent)
    {
        if (_stopRequested) return false;
        var target = _config?.TargetWindowHandle ?? IntPtr.Zero;
        if (target == IntPtr.Zero || !KeyboardSender.IsForeground(target))
        {
            Toast("warning", "游戏窗口已失去前台焦点，自动演奏已停止并释放按键。");
            return false;
        }
        if (!KeyboardSender.TrySendBatch(new[] { new KeyAction(0, inputEvent.ScanCode, inputEvent.IsDown) }))
        {
            Toast("error", "向游戏发送按键失败，自动演奏已停止并释放按键。");
            return false;
        }
        lock (_autoHeldLock)
        {
            if (inputEvent.IsDown) _autoHeldScanCodes.Add(inputEvent.ScanCode);
            else _autoHeldScanCodes.Remove(inputEvent.ScanCode);
        }
        _bridge?.Publish(new { type = "input.feedback", payload = new { lane = inputEvent.GameKey, pressed = inputEvent.IsDown } });
        return true;
    }

    private void ReleaseShowcaseKeys()
    {
        ushort[] held;
        lock (_autoHeldLock) { held = _autoHeldScanCodes.ToArray(); _autoHeldScanCodes.Clear(); }
        if (held.Length > 0)
            KeyboardSender.TrySendBatch(held.Select(scanCode => new KeyAction(0, scanCode, false)).ToArray());
        foreach (var scanCode in held)
        {
            int lane = _inputObserver.GameLaneBindings
                .Select((binding, index) => (binding, index))
                .Where(item => item.binding.ScanCode == scanCode)
                .Select(item => item.index + 1)
                .FirstOrDefault();
            if (lane != 0) _bridge?.Publish(new { type = "input.feedback", payload = new { lane, pressed = false } });
        }
    }

    private void HandlePracticeKey(InputObservation obs)
    {
        if (_practiceSession == null) return;
        var targetGroup = _practiceSession.GetTargetsAt(_practiceTimeline.CurrentTimeUs(obs.WallClockUs));
        if (!_practiceTimeline.IsRunning)
        {
            var upcoming = _practiceSession.GetUpcomingTargets(_currentTimeUs);
            if (upcoming.Any(n => n.GameKey == obs.GameKey))
            {
                _practiceDriverKey = obs.GameKey;
                _practiceTimeline.Resume(obs.WallClockUs);
                targetGroup = _practiceSession.GetTargetsAt(_currentTimeUs);
            }
        }

        long actualTimeUs = _practiceTimeline.CurrentTimeUs(obs.WallClockUs);
        var feedback = _practiceSession.EvaluateInput(obs.GameKey, actualTimeUs);
        if (feedback.Kind != "correct")
        {
            int expected = targetGroup.FirstOrDefault()?.GameKey ?? _practiceSession.GetUpcomingTargets(_currentTimeUs).FirstOrDefault()?.GameKey ?? 0;
            PublishPracticeState(feedback.Kind == "early" ? "当前时间没有待练音符" : $"按错了，当前需要 键 {expected}", false);
            return;
        }
        if (!_immersionState.IsOwnedByAutomaticPlayback) StartAutomaticImmersion();
        var target = feedback.TargetNoteId is Guid id ? _currentChart?.Notes.FirstOrDefault(n => n.Id == id) : null;
        if (target != null)
        {
            _judgedNotes.TryAdd(target.Id, 0);
            PublishNoteFeedback(target, HitGrade.Perfect);
            var nextPosition = _practiceSession.GetUpcomingTargets(actualTimeUs).FirstOrDefault()?.Position;
            if (nextPosition is { } next && target.Position is { } current
                && (next.Page != current.Page || next.Slot != current.Slot))
                QueuePositionTo(next);
        }
        _currentTimeUs = actualTimeUs;
        PublishPracticeState("正确", true);
    }

    private void HandleRhythmKey(InputObservation obs)
    {
        var actualTimeUs = _rhythmSession.ToTimelineTimeUs(obs.WallClockUs, _speedPercent);
        long inputWindowUs = 160_000L * _speedPercent / 100;
        var notes = _currentChart?.Notes;
        if (notes == null || notes.Count == 0) return;
        long earliestUs = actualTimeUs - inputWindowUs;
        long latestUs = actualTimeUs + inputWindowUs;
        int low = 0, high = notes.Count;
        while (low < high)
        {
            int mid = low + ((high - low) >> 1);
            if (notes[mid].StartUs < earliestUs) low = mid + 1;
            else high = mid;
        }
        Nte.GuitarChart.Core.ChartNote? note = null;
        long bestError = long.MaxValue;
        for (int i = low; i < notes.Count && notes[i].StartUs <= latestUs; i++)
        {
            var candidate = notes[i];
            if (candidate.GameKey != obs.GameKey || _judgedNotes.ContainsKey(candidate.Id)) continue;
            long error = Math.Abs(candidate.StartUs - actualTimeUs);
            if (error >= bestError) continue;
            note = candidate;
            bestError = error;
        }
        if (note == null) return;
        _judgedNotes.TryAdd(note.Id, 0);
        var grade = _rhythmSession.JudgeInput(note.StartUs, actualTimeUs);
        PublishNoteFeedback(note, grade);
    }

    private void PublishNoteFeedback(Nte.GuitarChart.Core.ChartNote note, HitGrade grade) =>

                noteId = note.Id,
                gameKey = note.GameKey,
                noteNumber = note.NoteNumber,
                velocity = note.Velocity,
                durationUs = note.DurationUs,

            }
        });

    private void StartSimulatedMidiPlayback(long startOffsetUs)
    {
        var chart = _currentChart;
        if (chart == null) return;
        double speedRatio = _speedPercent / 100d;
        var notes = chart.Notes.Select(note => new MidiAutoPlayer.Models.RawMidiNote

            StartUs = (long)(note.StartUs / speedRatio),

            TrackId = note.TrackIndex,
            Program = note.Program
        }).ToArray();

            notes,
            (long)(startOffsetUs / speedRatio),
            (long)(chart.DurationUs / speedRatio));
    }

    private void OnPositionSwitchKeyPressed(long wallClockUs)
    {
        long inputTimeUs = _isPlaying ? _rhythmSession.ToTimelineTimeUs(wallClockUs, _speedPercent) : _currentTimeUs;
        if (_isPlaying && _currentMode == PlayMode.Practice && _manualPracticePosition)
        {
            var practiceSwitch = _currentChart?.PositionEvents.FirstOrDefault(e => inputTimeUs >= e.PromptTimeUs && inputTimeUs <= e.DeadlineTimeUs);
            if (practiceSwitch != null)
            {
                _bridge?.Publish(new { type = "position.active", payload = new { page = practiceSwitch.To.Page, slot = practiceSwitch.To.Slot, stateText = $"已手动确认 P{practiceSwitch.To.Page} · S{practiceSwitch.To.Slot}" } });
                _judgedSwitches.TryAdd(practiceSwitch.Id, 0);
            }
            return;
        }
        if (!_isPlaying || _currentMode != PlayMode.RhythmHell || _executionMode != PositionExecutionMode.FullManual) return;
        var activeEvent = _currentChart?.PositionEvents.FirstOrDefault(e => inputTimeUs >= e.PromptTimeUs && inputTimeUs <= e.DeadlineTimeUs);
        if (activeEvent == null || !_judgedSwitches.TryAdd(activeEvent.Id, 0)) return;
        Toast("success", $"换位成功：P{activeEvent.To.Page} · S{activeEvent.To.Slot}");
    }

    private void OnGlobalKeyStateChanged(int virtualKey, int scanCode, bool isDown)
    {
        int gameLane = NativeInputObserver.MapConfiguredKeyToGameLane(virtualKey, scanCode, _inputObserver.GameLaneBindings);
        if (gameLane is >= 1 and <= 6)
            _bridge?.Publish(new { type = "input.feedback", payload = new { lane = gameLane, pressed = isDown } });

        if (_inputObserver.MatchesPositionSwitchKey(virtualKey, scanCode))
            _bridge?.Publish(new { type = "input.feedback", payload = new { lane = 7, pressed = isDown } });
    }

    private void PublishPracticeState(string message, bool? lastInputCorrect = null)
    {
        var targets = _practiceSession?.GetTargetsAt(_currentTimeUs) ?? Array.Empty<Nte.GuitarChart.Core.ChartNote>();
        if (targets.Count == 0 && _practiceSession != null) targets = _practiceSession.GetUpcomingTargets(_currentTimeUs);
        var target = targets.FirstOrDefault();
        var requiredKeys = targets.Where(n => n.GameKey.HasValue).Select(n => n.GameKey!.Value).Distinct().Order().ToArray();
        _bridge?.Publish(new
        {
            type = "practice.state",
            payload = new
            {
                mode = "practice",
                currentPage = target?.Position?.Page ?? 0,
                currentSlot = target?.Position?.Slot ?? 0,
                currentRequiredKeys = requiredKeys,
                playing = _isPlaying,
                timelineRunning = _practiceTimeline.IsRunning,
                currentTimeUs = _currentTimeUs,
                message
            }
        });
    }

    private void PublishPlaybackTick(string text, bool playing)
    {
        long nowMs = Environment.TickCount64;
        long minimumIntervalMs = _currentMode == PlayMode.Practice ? 40 : 250;
        if (playing && _lastPublishedPlaying == true && nowMs - Interlocked.Read(ref _lastPublishedTickMs) < minimumIntervalMs) return;
        Interlocked.Exchange(ref _lastPublishedTickMs, nowMs);
        _lastPublishedPlaying = playing;
        _bridge?.Publish(new { type = "playback.tick", payload = new { currentTimeUs = _currentTimeUs, durationUs = _currentChart?.DurationUs ?? 0, playing } });
    }

    private void PracticeTimelineLoop()
    {
        var chart = _currentChart;
        if (chart == null) return;
        while (!_stopRequested && _isPlaying && _currentMode == PlayMode.Practice)
        {
            long nowUs = MonotonicTimeUs();
            _currentTimeUs = _practiceTimeline.CurrentTimeUs(nowUs);
            PublishPlaybackTick(_practiceTimeline.IsRunning ? "练习进行中" : "等待按住目标键", _practiceTimeline.IsRunning);
            if (_practiceTimeline.IsRunning && _currentTimeUs >= chart.DurationUs)
            {
                _practiceTimeline.Pause(nowUs);
                _practiceDriverKey = 0;
                long? recoveryTimeUs = _practiceSession?.GetTailRecoveryTimeUs(_currentTimeUs, chart.DurationUs);
                if (recoveryTimeUs.HasValue)
                {
                    _currentTimeUs = recoveryTimeUs.Value;
                    _practiceTimeline.Seek(_currentTimeUs, nowUs);
                }
                _isPlaying = false;
                EndAutomaticImmersion();
                foreach (int lane in _heldGameKeys.Keys)
                    _bridge?.Publish(new { type = "input.feedback", payload = new { lane, pressed = false } });
                _heldGameKeys.Clear();
                _inputObserver.Stop();
                string message = recoveryTimeUs.HasValue
                    ? "已到曲尾，剩余目标已回到可练位置，按住目标键继续"
                    : "本段完成";
                PublishPracticeState(message);
                PublishPlaybackTick(message, false);
                return;
            }
            Thread.Sleep(10);
        }
    }

    private static long MonotonicTimeUs() =>
        System.Diagnostics.Stopwatch.GetTimestamp() * 1_000_000L / System.Diagnostics.Stopwatch.Frequency;
    private void Toast(string level, string message) => _bridge?.Publish(new { type = "app.toast", payload = new { level, message } });

    public void Dispose()
    {
        StopPlayback();
        _positionThreadStop = true;
        _positionSignal.Set();
        if (_positionThread.IsAlive && _positionThread != Thread.CurrentThread) _positionThread.Join(500);
        _inputObserver.Dispose();
        _simulatedMidiPlayer.Dispose();
        _window.Dispose();
        _pauseEvent.Dispose();
        _positionSignal.Dispose();
    }
```

```csharp
using System.Diagnostics;
using System.Runtime.InteropServices;

namespace MidiAutoPlayer.Input;

/// <summary>
/// Observes one global key without consuming it, so the foreground application still receives it.
/// </summary>
public sealed class GlobalKeyObserver : IDisposable
{
    private const int WhKeyboardLl = 13;
    private const int WmKeyDown = 0x0100;
    private const int WmKeyUp = 0x0101;
    private const int WmSysKeyDown = 0x0104;
    private const int WmSysKeyUp = 0x0105;

    [StructLayout(LayoutKind.Sequential)]
    private struct KeyboardHookData
    {
        public uint VirtualKey;
        public uint ScanCode;
        public uint Flags;
        public uint Time;
        public IntPtr ExtraInfo;
    }

    private delegate IntPtr LowLevelKeyboardProc(int code, IntPtr message, IntPtr data);

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern IntPtr SetWindowsHookEx(int hookId, LowLevelKeyboardProc callback, IntPtr module, uint threadId);

    [DllImport("user32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool UnhookWindowsHookEx(IntPtr hook);

    [DllImport("user32.dll", SetLastError = true)]
    private static extern IntPtr CallNextHookEx(IntPtr hook, int code, IntPtr message, IntPtr data);

    [DllImport("kernel32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern IntPtr GetModuleHandle(string? moduleName);

    private readonly LowLevelKeyboardProc _callback;
    private IntPtr _hook;
    private int _virtualKey;
    private int _isDown;

    public event Action<int>? KeyPressed;

    public GlobalKeyObserver(int virtualKey)
    {
        _virtualKey = virtualKey;
        _callback = HookCallback;
    }

    public bool Start()
    {
        if (_hook != IntPtr.Zero) return true;
        using var process = Process.GetCurrentProcess();
        using var module = process.MainModule;
        _hook = SetWindowsHookEx(WhKeyboardLl, _callback, GetModuleHandle(module?.ModuleName), 0);
        return _hook != IntPtr.Zero;
    }

    public void SetVirtualKey(int virtualKey)
    {
        Interlocked.Exchange(ref _virtualKey, virtualKey);
        Interlocked.Exchange(ref _isDown, 0);
    }

    private IntPtr HookCallback(int code, IntPtr message, IntPtr data)
    {
        if (code >= 0)
        {
            int messageId = message.ToInt32();
            bool isDown = messageId is WmKeyDown or WmSysKeyDown;
            bool isUp = messageId is WmKeyUp or WmSysKeyUp;
            if (isDown || isUp)
            {
                int virtualKey = (int)Marshal.PtrToStructure<KeyboardHookData>(data).VirtualKey;
                if (virtualKey == Volatile.Read(ref _virtualKey))
                {
                    if (isUp)
                        Interlocked.Exchange(ref _isDown, 0);
                    else if (Interlocked.Exchange(ref _isDown, 1) == 0)
                        KeyPressed?.Invoke(virtualKey);
                }
            }
        }

        return CallNextHookEx(_hook, code, message, data);
    }

    public void Dispose()
    {
        if (_hook == IntPtr.Zero) return;
        UnhookWindowsHookEx(_hook);
        _hook = IntPtr.Zero;
    }
}
```

```csharp
    private const uint KEYEVENTF_EXTENDEDKEY = 0x0001;
    private const uint KEYEVENTF_KEYUP = 0x0002;
    private const uint KEYEVENTF_SCANCODE = 0x0008;
    private const uint MAPVK_VSC_TO_VK_EX = 3;
    private const uint WM_KEYDOWN = 0x0100;
    private const uint WM_KEYUP = 0x0101;

        public MOUSEKEYBDHARDWAREINPUT mkhi;

        [FieldOffset(0)]

    [DllImport("user32.dll", CharSet = CharSet.Auto)]
    private static extern uint MapVirtualKey(uint code, uint mapType);

    [DllImport("user32.dll", SetLastError = true)]
    private static extern bool PostMessage(IntPtr hWnd, uint msg, IntPtr wParam, IntPtr lParam);

    [DllImport("user32.dll")]
    public static extern IntPtr GetForegroundWindow();

    [DllImport("user32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern int GetWindowText(IntPtr hWnd, StringBuilder lpString, int nMaxCount);

    public static string GetWindowTitle(IntPtr hWnd)
    {
        if (hWnd == IntPtr.Zero) return "(None)";
        var sb = new StringBuilder(256);
        if (GetWindowText(hWnd, sb, sb.Capacity) > 0)
            return sb.ToString();
        return "(Unnamed Window)";
    }

    public static bool IsForegroundWindow(IntPtr hWnd)
    {
        if (hWnd == IntPtr.Zero) return false;
        return GetForegroundWindow() == hWnd;
    }

    public static bool IsForeground(IntPtr hWnd) => IsForegroundWindow(hWnd);

    public static void SendBatch(IReadOnlyList<KeyAction> actions)
    {
        TrySendBatch(actions);
    }

    public static bool TrySendBatch(IReadOnlyList<KeyAction> actions)
    {
        if (actions == null || actions.Count == 0) return true;

        var inputs = new INPUT[actions.Count];
        int structSize = Marshal.SizeOf<INPUT>();

        for (int i = 0; i < actions.Count; i++)
        {
            var act = actions[i];
            uint flags = KEYEVENTF_SCANCODE;
            if (!act.IsDown) flags |= KEYEVENTF_KEYUP;
            if (act.Extended) flags |= KEYEVENTF_EXTENDEDKEY;

            inputs[i] = new INPUT
            {
                type = INPUT_KEYBOARD,
                mkhi = new MOUSEKEYBDHARDWAREINPUT
                {
                    ki = new KEYBDINPUT
                    {
                        wVk = 0,
                        wScan = act.ScanCode,
                        dwFlags = flags,
                        time = 0,
                        dwExtraInfo = IntPtr.Zero
                    }
                }
            };
        }

        return SendInput((uint)inputs.Length, inputs, structSize) == (uint)inputs.Length;
    }

    public static bool TryPostBatch(IntPtr hWnd, IReadOnlyList<KeyAction> actions)
    {
        if (hWnd == IntPtr.Zero) return false;
        if (actions == null || actions.Count == 0) return true;

        foreach (KeyAction action in actions)
        {
            uint scanCode = action.ScanCode | (action.Extended ? 0xE000u : 0u);
            uint virtualKey = MapVirtualKey(scanCode, MAPVK_VSC_TO_VK_EX);
            if (virtualKey == 0) return false;

            uint lParam = 1u | ((uint)action.ScanCode << 16);
            if (action.Extended) lParam |= 1u << 24;
            if (!action.IsDown) lParam |= (1u << 30) | (1u << 31);

            uint message = action.IsDown ? WM_KEYDOWN : WM_KEYUP;
            if (!PostMessage(hWnd, message, (IntPtr)virtualKey, (IntPtr)(long)lParam))
                return false;
        }

        return true;
    }

    public static void ReleaseAllKeys(KeyPreset? preset = null)
    {
        var keysToRelease = new HashSet<(ushort ScanCode, bool Extended)>();

        if (preset != null)
        {
            foreach (var mapping in preset.Keys)
            {
                keysToRelease.Add((mapping.ScanCode, mapping.Extended));
            }
        }
        else
        {
            foreach (var item in PresetManager.DefaultMappingData)
            {
                keysToRelease.Add((item.ScanCode, item.Ext));
            }
        }

        var actions = new List<KeyAction>(keysToRelease.Count);
        foreach (var (scanCode, extended) in keysToRelease)
        {
            actions.Add(new KeyAction(0, scanCode, isDown: false, extended: extended));
        }

        SendBatch(actions);
    }
```
